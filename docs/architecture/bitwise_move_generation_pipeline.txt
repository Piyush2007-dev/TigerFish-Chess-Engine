  1. ray = ray_table[dir][king_sq]      │        ← 1 array lookup         │
  2. blockers = ray & occupancy[2]      │        ← 1 AND                  │

  9. pinned |= first_bb                 │        ← 1 OR                   │
 10. pin_mask[first] = ray | back_ray   │        ← 1 OR + write           │

  ┌──────────────────────────────────────────────────────────────────────┐
  │  pawnatk(board, checkers)           — Enemy PAWN attacks            │
  │    Per enemy pawn:                                                  │
  │      left_side  = (lsb & NOT_A_FILE) >> 9    ← 1 AND + 1 SHIFT     │
  │      right_side = (lsb & NOT_H_FILE) >> 7    ← 1 AND + 1 SHIFT     │
  │      attacks    = left_side | right_side       ← 1 OR               │
  │      if attacks & king_bb → checkers |= lsb   ← 1 AND + 1 OR       │
  │      attack_mask |= attacks                   ← 1 OR               │
  │      mask &= mask-1   (clear LSB)             ← 1 SUB + 1 AND      │
  │  Per pawn: ~8 ops.   Up to 8 enemy pawns → max ~64 ops             │
  └──────────────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────────────┐
  │  knightatk(board, checkers)         — Enemy KNIGHT attacks          │
  │    Per enemy knight:                                                │
  │      l1 = (lsb & NOT_H_FILE)  << 1   ← AND + SHIFT                │
  │      l2 = (lsb & NOT_GH_FILE) << 2   ← AND + SHIFT                │
  │      r1 = (lsb & NOT_A_FILE)  >> 1   ← AND + SHIFT                │
  │      r2 = (lsb & NOT_AB_FILE) >> 2   ← AND + SHIFT                │
  │      h1 = l1 | r1                    ← OR                          │
  │      h2 = l2 | r2                    ← OR                          │
  │      attacks = (h1<<16)|(h1>>16)|(h2<<8)|(h2>>8) ← 4 SHIFTS + 3 OR│
  │      if attacks & king_bb → checkers |= lsb  ← AND + OR            │
  │      attack_mask |= attacks          ← OR                          │
  │      mask &= mask-1                  ← SUB + AND                   │
  │  Per knight: ~14 ops.  Up to 10 enemy knights → max ~140 ops       │
  └──────────────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────────────┐
  │  sliding_atks(board, BISHOP, 4,8, checkers)   — Enemy BISHOP atks  │
  │  sliding_atks(board, ROOK,   0,4, checkers)   — Enemy ROOK atks    │
  │  sliding_atks(board, QUEEN,  0,8, checkers)   — Enemy QUEEN atks   │
  │                                                                     │
  │    occ = occupancy[2] ^ king_bitboard   ← XOR (king transparent)   │
  │    Per piece:                                                       │
  │      bishop_attacks(sq, occ):                                       │

  │              ← 1 MUL + 1 SHIFT                                     │
  │        return BISHOP_ATTACKS[sq][idx]   ← 1 table lookup           │
  │      Total per bishop: 4 ops                                        │
  │      if attacks & king_bb → checkers |= (1ULL<<sq)  ← AND + OR    │

  │      mask &= mask-1                     ← SUB + AND                │
  │  Per sliding piece: ~7 ops. Up to 9 sliders → max ~63 ops          │

  │  kingatk(board)                     — Enemy KING attacks            │

╔══════════════════════════════════════════════════════════════════════════╗
║  PHASE 3: MOVE GENERATION PER PIECE TYPE (6 functions)                 ║
╚══════════════════════════════════════════════════════════════════════════╝

──────────────────────────────────────────────────────
  KingMoves(board, moves, attack_mask)
──────────────────────────────────────────────────────
  east  = (king & NOT_H_FILE) << 1       ← AND + SHIFT
  west  = (king & NOT_A_FILE) >> 1       ← AND + SHIFT
  r     = king | east | west             ← 2 OR
  raw   = (r<<8 | r | r>>8) & ~lsb      ← 2 SHIFT + 2 OR + NOT + AND
  raw  &= ~attack_mask                   ← NOT + AND  (remove attacked sqs)
  captures = raw & occupancy[enemy]      ← AND
  moves    = raw & ~occupancy[2]         ← NOT + AND
  Castling: check castling_rights bits,  ← 2 AND + 2 AND bitmask checks
            occupancy[2] clear squares,
            attack_mask clear king path
  Per square found: pack_move() →32-bit  ← 5 shifts + 5 ORs per move
  ~12 ops base + ~10 per legal square

──────────────────────────────────────────────────────
  PawnMoves(board, moves, pins, legal_mask)
──────────────────────────────────────────────────────
  single_push  = (pawn << 8) & ~occupancy[2]               ← SHIFT + NOT + AND
  rank2_pawns  = pawn & RANK_2                              ← AND
  first_step   = (rank2_pawns << 8) & ~occupancy[2]         ← SHIFT + NOT + AND
  double_push  = (first_step << 8) & ~occupancy[2]          ← SHIFT + NOT + AND
  left_caps    = (pawn & NOT_H_FILE) << 9                   ← AND + SHIFT
  right_caps   = (pawn & NOT_A_FILE) << 7                   ← AND + SHIFT
  captures     = (left_caps | right_caps) & occupancy[enemy]← OR + AND
  captures    &= legal_mask                                 ← AND
  push_mask    = (single|double) & legal_mask & ~promo_rank ← AND + AND + NOT
  Per legal target square:
    check pin_mask[from] & (1ULL<<to)    ← 1 AND
    pack_move()                          ← 5 shifts + 5 ORs
  En passant special:
    ep_bb = 1ULL << en_passant           ← SHIFT
    ep_pawns check 2 diagonal sqs        ← 4 AND + 2 SHIFT + 2 OR
    ep_exposes_king() if needed          ← ~40 more ops (8 ray checks)
  ~20 base ops + ~12 per legal pawn

──────────────────────────────────────────────────────
  KnightMoves → append_knight_moves()
──────────────────────────────────────────────────────
  Per knight:
    if (pinned & lsb): skip             ← 1 AND (pinned knight = 0 moves)
    knight_attacks() → 14 ops (as above)
    moves_mask &= legal_mask            ← AND
    captures   &= legal_mask            ← AND
    pack_move() per legal square        ← 5 shifts + 5 ORs per move
  ~16 base ops per knight

──────────────────────────────────────────────────────
  BishopMoves  → append_sliding_moves(piece, dir 4→8)
  RookMoves    → append_sliding_moves(piece, dir 0→4)
  QueenMoves   → append_sliding_moves(piece, dir 0→8)
──────────────────────────────────────────────────────
  Per piece:
    pmask = pins.pin_mask[from_sq]      ← 1 array lookup
    bishop/rook/queen_attacks(sq, occ): ← 4 ops (MAGIC lookup)
    attacks &= ~own_occ                 ← NOT + AND
    attack_mask |= attacks              ← OR
    legal = attacks & legal_mask & pmask← 2 AND
    quiets   = legal & ~all_occ        ← NOT + AND
    captures = legal & enemy_occ       ← AND
    Per quiet: pop_lsb + pack_move     ← BSF + SUB + AND + 10 shift/OR ops
    Per capture: same + piece_on lookup← 12 ops
    mask &= mask-1  (advance to next)  ← SUB + AND
  ~12 base ops per sliding piece

╔══════════════════════════════════════════════════════════════════════════╗
║  PHASE 4: PACK EACH LEGAL MOVE INTO 32-BIT INTEGER                     ║
║  pack_move(from, to, piece, cap, promo, castle, ep, dbl_push)          ║
╚══════════════════════════════════════════════════════════════════════════╝
  m  = from & 0x3F              Bits 0–5   ← AND
  m |= (to & 0x3F) << 6         Bits 6–11  ← AND + SHIFT + OR
  m |= (piece & 0xF) << 12      Bits 12–15 ← AND + SHIFT + OR
  m |= (cap & 0xF) << 16        Bits 16–19 ← AND + SHIFT + OR
  m |= (promo & 0xF) << 20      Bits 20–23 ← AND + SHIFT + OR
  m |= (castling_rights) << 24  Bits 24–27 ← SHIFT + OR
  m |= is_castle << 28          Bit 28     ← SHIFT + OR
  m |= is_ep << 29              Bit 29     ← SHIFT + OR
  m |= is_dbl_push << 30        Bit 30     ← SHIFT + OR
  m |= side_to_move << 31       Bit 31     ← SHIFT + OR
  Total: ~20 bitwise ops per move packed
